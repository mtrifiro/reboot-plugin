Below is a Reboot bank backend using a protobuf API: one file for the API, two servicers and an entry point. Reboot's Python API has changed between releases, so check the comments marked `NOTE` against the version you have installed.

### Layout
```
api/bank/v1/bank.proto
backend/src/account_servicer.py
backend/src/bank_servicer.py
backend/src/main.py
.rbtrc
```

### `api/bank/v1/bank.proto`
```proto
syntax = "proto3";

package bank.v1;

import "rbt/v1alpha1/options.proto";

// ---------------- Account ----------------

message Account {
  option (rbt.v1alpha1.state) = {};
  uint64 balance = 1;  // Unsigned: can never be negative.
}

service AccountMethods {
  rpc Open(OpenRequest) returns (OpenResponse) {
    option (rbt.v1alpha1.method).writer = { constructor: {} };
  }
  rpc Balance(BalanceRequest) returns (BalanceResponse) {
    option (rbt.v1alpha1.method).reader = {};
  }
  rpc Deposit(DepositRequest) returns (DepositResponse) {
    option (rbt.v1alpha1.method) = {
      writer: {},
      errors: [ "InvalidAmountError" ],
    };
  }
  rpc Withdraw(WithdrawRequest) returns (WithdrawResponse) {
    option (rbt.v1alpha1.method) = {
      writer: {},
      errors: [ "InvalidAmountError", "OverdraftError" ],
    };
  }
}

message OpenRequest {}
message OpenResponse {}

message BalanceRequest {}
message BalanceResponse { uint64 balance = 1; }

message DepositRequest { uint64 amount = 1; }
message DepositResponse { uint64 balance = 1; }

message WithdrawRequest { uint64 amount = 1; }
message WithdrawResponse { uint64 balance = 1; }

message InvalidAmountError {}
message OverdraftError { uint64 shortfall = 1; }

// ---------------- Bank (singleton, coordinates transfers) ----------------

message Bank {
  option (rbt.v1alpha1.state) = {};
  repeated string account_ids = 1;
}

service BankMethods {
  rpc Create(CreateRequest) returns (CreateResponse) {
    option (rbt.v1alpha1.method).writer = { constructor: {} };
  }
  rpc OpenAccount(OpenAccountRequest) returns (OpenAccountResponse) {
    option (rbt.v1alpha1.method).transaction = {};
  }
  rpc Transfer(TransferRequest) returns (TransferResponse) {
    option (rbt.v1alpha1.method) = {
      transaction: {},
      errors: [ "InvalidTransferError" ],
    };
  }
}

message CreateRequest {}
message CreateResponse {}

message OpenAccountRequest { string account_id = 1; }
message OpenAccountResponse {}

message TransferRequest {
  string from_account_id = 1;
  string to_account_id = 2;
  uint64 amount = 3;
}
message TransferResponse {}

message InvalidTransferError { string reason = 1; }
```

### `backend/src/account_servicer.py`
```python
from bank.v1.bank_rbt import (
    Account,
    BalanceRequest, BalanceResponse,
    DepositRequest, DepositResponse,
    InvalidAmountError,
    OpenRequest, OpenResponse,
    OverdraftError,
    WithdrawRequest, WithdrawResponse,
)
from reboot.aio.auth.authorizers import allow
from reboot.aio.contexts import ReaderContext, WriterContext


class AccountServicer(Account.Servicer):

    def authorizer(self):
        # NOTE: allow-all for demo purposes; replace with real auth.
        return allow()

    async def open(
        self, context: WriterContext, request: OpenRequest
    ) -> OpenResponse:
        self.state.balance = 0
        return OpenResponse()

    async def balance(
        self, context: ReaderContext, request: BalanceRequest
    ) -> BalanceResponse:
        return BalanceResponse(balance=self.state.balance)

    async def deposit(
        self, context: WriterContext, request: DepositRequest
    ) -> DepositResponse:
        if request.amount == 0:
            raise Account.DepositAborted(InvalidAmountError())
        self.state.balance += request.amount
        return DepositResponse(balance=self.state.balance)

    async def withdraw(
        self, context: WriterContext, request: WithdrawRequest
    ) -> WithdrawResponse:
        if request.amount == 0:
            raise Account.WithdrawAborted(InvalidAmountError())
        # Reject overdrafts so the balance never goes below zero.
        if request.amount > self.state.balance:
            raise Account.WithdrawAborted(
                OverdraftError(shortfall=request.amount - self.state.balance)
            )
        self.state.balance -= request.amount
        return WithdrawResponse(balance=self.state.balance)
```

### `backend/src/bank_servicer.py`
```python
from bank.v1.bank_rbt import (
    Account,
    Bank,
    CreateRequest, CreateResponse,
    InvalidTransferError,
    OpenAccountRequest, OpenAccountResponse,
    TransferRequest, TransferResponse,
)
from reboot.aio.auth.authorizers import allow
from reboot.aio.contexts import TransactionContext, WriterContext


class BankServicer(Bank.Servicer):

    def authorizer(self):
        return allow()

    async def create(
        self, context: WriterContext, request: CreateRequest
    ) -> CreateResponse:
        return CreateResponse()

    async def open_account(
        self, context: TransactionContext, request: OpenAccountRequest
    ) -> OpenAccountResponse:
        # Construct the account and record it, atomically.
        await Account.open(context, request.account_id)
        self.state.account_ids.append(request.account_id)
        return OpenAccountResponse()

    async def transfer(
        self, context: TransactionContext, request: TransferRequest
    ) -> TransferResponse:
        if request.from_account_id == request.to_account_id:
            raise Bank.TransferAborted(
                InvalidTransferError(reason="Cannot transfer to same account")
            )

        # Both calls run inside one transaction: if the withdraw aborts
        # (e.g. overdraft) or the deposit fails, neither change commits.
        await Account.ref(request.from_account_id).withdraw(
            context, amount=request.amount
        )
        await Account.ref(request.to_account_id).deposit(
            context, amount=request.amount
        )
        return TransferResponse()
```

### `backend/src/main.py`
```python
import asyncio

from account_servicer import AccountServicer
from bank.v1.bank_rbt import Bank
from bank_servicer import BankServicer
from reboot.aio.applications import Application
from reboot.aio.external import InitializeContext

BANK_ID = "reboot-bank"


async def initialize(context: InitializeContext):
    # Create the singleton Bank on startup.
    # NOTE: initialize may re-run; constructors are idempotent per-ID in
    # recent Reboot versions, otherwise wrap with `Bank.idempotently()`.
    await Bank.create(context, BANK_ID)


async def main():
    await Application(
        servicers=[AccountServicer, BankServicer],
        initialize=initialize,
    ).run()


if __name__ == "__main__":
    asyncio.run(main())
```

### `.rbtrc`
```
generate api/ --python=backend/api/
dev run --python --application=backend/src/main.py --watch=backend/src/**/*.py
dev run --name=bank
```

### Notes
- **Non-negative balance:** `withdraw` rejects any amount larger than the balance. The unsigned `uint64` field adds a second guard.
- **Atomic transfer:** `Bank.transfer` is a Reboot `transaction`. If the withdraw raises `OverdraftError`, or the deposit fails, the whole transaction rolls back and no partial transfer is committed.
- **Version differences:** Older Reboot releases pass `state` as a method argument instead of `self.state`. The way constructors are called (`Account.open(context, id)`) has also changed between releases. Newer releases also offer a Pydantic `reboot.api` style as an alternative to `.proto` files.

Confidence: medium
