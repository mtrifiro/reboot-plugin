```proto
// bank/v1/account.proto
syntax = "proto3";

package bank.v1;

import "google/protobuf/empty.proto";
import "rbt/v1alpha1/options.proto";

// Persisted state for one account (amounts in cents).
message AccountState {
  option (rbt.v1alpha1.state) = {};
  int64 balance = 1;
}

message OpenRequest {}

message DepositRequest {
  int64 amount = 1;
}

message WithdrawRequest {
  int64 amount = 1;
}

message TransferRequest {
  string to_account_id = 1;
  int64 amount = 2;
}

message BalanceResponse {
  int64 balance = 1;
}

// Declared error returned when a withdrawal would overdraw the account.
message InsufficientFunds {
  int64 balance = 1;
  int64 requested = 2;
}

service Account {
  rpc Open(OpenRequest) returns (google.protobuf.Empty) {
    option (rbt.v1alpha1.method).constructor = {};
  }
  rpc Deposit(DepositRequest) returns (BalanceResponse) {
    option (rbt.v1alpha1.method).writer = {};
  }
  rpc Withdraw(WithdrawRequest) returns (BalanceResponse) {
    option (rbt.v1alpha1.method).writer = {};
  }
  // Atomic across both accounts: all or nothing.
  rpc Transfer(TransferRequest) returns (BalanceResponse) {
    option (rbt.v1alpha1.method).transaction = {};
  }
}
```

```python
class AccountServicerImpl(AccountServicer):

        self,

        state: AccountState,
        request: OpenRequest,
    ) -> Empty:
        # New accounts start at zero.
        state.balance = 0
        return Empty()

    async def deposit(
        self,
        context: WriterContext,
        state: AccountState,
        request: DepositRequest,
    ) -> BalanceResponse:
        if request.amount <= 0:
            raise ValueError("deposit amount must be positive")
        state.balance += request.amount
        return BalanceResponse(balance=state.balance)

    async def withdraw(
        self,
        context: WriterContext,
        state: AccountState,
        request: WithdrawRequest,
    ) -> BalanceResponse:
        if request.amount <= 0:
            raise ValueError("withdraw amount must be positive")
        # Enforce the non-negative balance invariant.
        if request.amount > state.balance:
            raise InsufficientFunds(
                balance=state.balance,
                requested=request.amount,
            )
        state.balance -= request.amount
        return BalanceResponse(balance=state.balance)

    async def transfer(
        self,
        context: TransactionContext,
        state: AccountState,
        request: TransferRequest,
    ) -> BalanceResponse:
        if request.amount <= 0:
            raise ValueError("transfer amount must be positive")
        # Either both calls succeed or the transaction rolls back.
        # Withdraw raises InsufficientFunds, aborting the transfer.
        await Account.ref(context.state_id).withdraw(
            context, amount=request.amount
        )
        await Account.ref(request.to_account_id).deposit(
            context, amount=request.amount
        )
        balance = (await Account.ref(context.state_id).withdraw_balance(context)).balance
        return BalanceResponse(balance=balance)


async def main() -> None:
    await Application(servicers=[AccountServicerImpl]).run()

    asyncio.run(main())
```
